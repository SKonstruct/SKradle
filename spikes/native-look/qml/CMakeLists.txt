cmake_minimum_required(VERSION 3.21)
project(nl_qml LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(nl_qml main.cpp)
qt_add_qml_module(nl_qml URI Spike VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(nl_qml PRIVATE Qt6::Quick Qt6::QuickControls2)
